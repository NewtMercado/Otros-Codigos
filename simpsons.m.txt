function integral = simpsons_rule(f, a, b, n)
    % Método de Simpson para calcular la integral de una función
    % f: función a integrar
    % a, b: límites de integración
    % n: número de subintervalos (debe ser par)

    % Paso de integración
    h = (b - a) / n;

    % Calcular los puntos de evaluación y evaluar la función
    x = linspace(a, b, n+1);
    y = f(x);

    % Aplicar la fórmula de Simpson
    integral = h/3 * (y(1) + 4*sum(y(2:2:end-1)) + 2*sum(y(3:2:end-2)) + y(end));
end
